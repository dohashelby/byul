def require(ok,code):
 if not ok:raise ValueError(code)

def same_packet(runtime):
 require(runtime.packet is runtime.driver.PACKET,'SELECTED_SESSION_SAME_DRIVER_PACKET_REQUIRED')

@contextmanager
def qualified_runtime(driver):
 """Consume the original driver's actual checked-file loader and packet object."""
 names=('s2_selected_session_fixed_core','s2_selected_session_fixed_stream')
 missing=object();saved={n:sys.modules.get(n,missing) for n in names}
 try:
  modules=[];raws=[]
  for name,filename,expected in zip(names,('session_core.py','gb12_stream.py'),(CORE,STREAM)):
   pin=driver.PACKET['local_files'][filename]
   require((pin['bytes'],pin['sha256'])==expected,'SELECTED_SESSION_FIXED_DEPENDENCY_PIN')
   raw=driver.checked_file(filename,pin);require((len(raw),sha(raw))==expected,'SELECTED_SESSION_FIXED_DEPENDENCY_BYTES')
   m=ModuleType(name);m.__file__=str(driver.OUT/filename);sys.modules[name]=m
   exec(compile(raw,m.__file__,'exec'),m.__dict__);modules.append(m);raws.append(raw)
  core,stream=modules
  runtime=SimpleNamespace(driver=driver,adapter=driver.boto3,packet=driver.PACKET,root=driver.OUT)
  same_packet(runtime)
  recovery=core.Recovery(runtime,stream,raws[1],driver.PACKET['local_files']['gb12_stream.py'])
  require(recovery.rt is runtime and recovery.driver is driver and recovery.rt.packet is driver.PACKET,'SELECTED_SESSION_QUALIFIED_CONSTRUCTOR_IDENTITY')
  require(core.STREAM_CAP==STREAM_CAP and stream.CHUNK==12000,'SELECTED_SESSION_ORIGINAL_CAP_CHUNK_REQUIRED')
  yield runtime,core,stream,raws[1],recovery
 finally:
  for n,prior in saved.items():
   if prior is missing:sys.modules.pop(n,None)
   else:sys.modules[n]=prior

def wire_upper_bound(size):
 require(type(size) is int and size>0,'SELECTED_SESSION_POSITIVE_INTEGER_BYTES')
 # D frame overhead <=160 bytes for fixed UUIDs, 12k chunks and <=512MiB archive.
 # H/F+remote terminal+all banner allowance remain bounded below this reserve.
 frames=(size+11999)//12000
 return 4*((size+2)//3)+frames*160+8192+65536

def positive_binding(driver,binding,whole_submission,whole_terminal,probe_submission,probe_terminal):
 packet=driver.PACKET;scope=packet['selected_archive_session_scope']
 expected=scope['remote_path'];name=scope['remote_name']
 require(scope['instance_id']==driver.IID==IID and scope['GET_fallback'] is False,'SELECTED_SESSION_EXACT_TARGET_NO_FALLBACK')
 require(type(name) is str and re.fullmatch(r'GB[0-9]{2,}-original-evidence\.zip',name) is not None
  and expected==BASE+name and binding.ARCHIVE==expected,'SELECTED_SESSION_FROZEN_CURRENT_ARCHIVE_PATH')
 prefix=packet['whole_argv_prefix'];actual=whole_submission['actual_argv']
 require(type(actual) is list and actual[:len(prefix)]==prefix,'SELECTED_SESSION_CURRENT_WHOLE_PREFIX')
 selected=binding.select_archive(whole_submission,whole_terminal,probe_submission,probe_terminal,
  expected_whole_argv=actual,expected_probe_argv=packet['probe_argv'],configuration_sha256=packet['configuration_sha256'])
 if selected is None:return None,None
 require(selected['remote_path']==expected and selected['remote_name']==name,'SELECTED_SESSION_DISCOVERY_SCOPE')
 require(selected['product_complete_claimed'] is False and wire_upper_bound(selected['bytes'])<=STREAM_CAP,'SELECTED_SESSION_ORIGINAL24MIB_WIRE_CAP')
 # select_archive already validated this exact wire's positive current wrapper.
 value=json.loads(binding.unpack(probe_terminal['StandardOutputContent']))
 wrapper=value['wrapper'];boot=wrapper['boot_id']
 require(str(uuid.UUID(boot))==boot and uuid.UUID(boot).int!=0,'SELECTED_SESSION_ACTUAL_WRAPPER_BOOT')
 require(selected['whole_command_id']==whole_terminal['CommandId'] and selected['probe_command_id']==probe_terminal['CommandId'],'SELECTED_SESSION_CURRENT_CID_JOIN')
 return selected,boot

def receive_bound(runtime,core,stream,source,recovery,selected,current_boot,output):
 same_packet(runtime);driver=runtime.driver;nonce=str(uuid.uuid4())

 core.require(driver.states()[IID]=='running' and not driver.active_commands(),'SELECTED_RUNNING_IDLE_REQUIRED')
 core.require(core.boot()==host_boot,'HOST_BOOT_CHANGED')

   'request_sha256':core.sha(core.jb(recovery.request)),'actual_STS_owner_values':recovery.owners,'original_end':end,'host_boot':host_boot,
   'current_archive_positive_binding':selected})
  recovery.session_submitted=True

  recovery.sid=recovery.transcript.sid

  core.require(recovery.server_terminal and row['Status']=='Terminated','POSITIVE_ORIGINAL_SESSION_TERMINATED_REQUIRED')

def collect(driver,binding,whole_submission,whole_terminal,probe_submission,probe_terminal):
 selected,current_boot=positive_binding(driver,binding,whole_submission,whole_terminal,probe_submission,probe_terminal)
 driver.put('exact-new-archive-binding.json',{'at':driver.now(),'selected':selected,'static_old_archive_lookup_used':False,
  'collector':'qualified-current-case-Session','actual_wrapper_boot':current_boot})
 if selected is None:return False
 folder=driver.OUT/'actual-original';folder.mkdir(mode=0o700)
 output=folder/selected['remote_name'];original=None;result=None
 with qualified_runtime(driver) as (runtime,core,stream,source,recovery):
  try:result=receive_bound(runtime,core,stream,source,recovery,selected,current_boot,output)
  except BaseException as error:original=error
  try:recovery.reconcile_cleanup_once()
  except BaseException as cleanup:
   if original is None:original=cleanup
   else:
    original.mutation_uncertain=True
    original.add_note('Independent own-Session cleanup: '+type(cleanup).__name__+': '+str(cleanup))
  if original is not None:raise original.with_traceback(original.__traceback__)
 require(result['bytes']==selected['bytes'] and result['sha256']==selected['sha256'],'SELECTED_SESSION_EXACT_RECEIVER_RESULT')
 # Preserve the original final exact-byte guard after transport success.
 require(output.stat().st_size==selected['bytes'] and sha(output.read_bytes())==selected['sha256'],'SELECTED_SESSION_LOCAL_ARCHIVE_INTEGRITY')
 driver.put('exact-new-archive-recovered.json',{'at':driver.now(),'binding':selected,'local':str(output),
  'exact_bytes_SHA_verified':True,'inner_CRC_or_semantics_verified':False,
  'normal_stop_owner':'original-driver-finally-only','GET_fallback_used':False,'product_success':False})
 return True
