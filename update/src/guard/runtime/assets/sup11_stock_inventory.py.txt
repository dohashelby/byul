"""Describe the approved mounted stock model without importing model or engine code."""

import hashlib
import json
import os
import stat
import sys


CONFIG_NAMES = ('config.json', 'preprocessor_config.json', 'tokenizer_config.json')


def has_auto_map(value):
    if isinstance(value, dict):
        return 'auto_map' in value or any(has_auto_map(item) for item in value.values())
    return isinstance(value, list) and any(has_auto_map(item) for item in value)


def config_observation(root, name):
    with os.fdopen(os.open(name, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK, dir_fd=root), 'rb') as stream:
        info = os.fstat(stream.fileno())
        if not stat.S_ISREG(info.st_mode) or not 0 < info.st_size <= 65536:
            raise ValueError()
        data = stream.read(info.st_size + 1)
        if len(data) != info.st_size:
            raise ValueError()
    document = json.loads(data)
    if not isinstance(document, dict) or has_auto_map(document):
        raise ValueError()
    return hashlib.sha256(data).hexdigest(), document


def observe():
    if not (sys.flags.isolated and sys.flags.no_site and sys.flags.dont_write_bytecode):
        raise ValueError()
    root = os.open('/models', os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)
    try:
        if not os.fstatvfs(root).f_flag & os.ST_RDONLY:
            raise ValueError()
        files = {}
        with os.scandir(root) as entries:
            for entry in entries:
                info = entry.stat(follow_symlinks=False)
                if len(files) >= 11 or not stat.S_ISREG(info.st_mode):
                    raise ValueError()
                files[entry.name] = info.st_size
        configs = {name: config_observation(root, name) for name in CONFIG_NAMES}
        model = configs['config.json'][1]
        processor = configs['preprocessor_config.json'][1]
        if (model.get('model_type') != 'qwen2_vl'
                or model.get('architectures') != ['Qwen2VLForConditionalGeneration']
                or processor.get('processor_class') != 'Qwen2VLProcessor'):
            raise ValueError()
    finally:
        os.close(root)
    with open('/proc/1/stat', 'rb') as stream:
        stat_line = stream.read(4096)
    rest = stat_line.rsplit(b') ', 1)[-1].split()
    return {'file_sizes': files, 'config_sha256': {name: row[0] for name, row in configs.items()},
            'read_only': True, 'auto_map_present': False, 'pid1_start_ticks': int(rest[19])}


try:
    print(json.dumps(observe(), sort_keys=True, separators=(',', ':')))
except (IndexError, OSError, RecursionError, TypeError, ValueError):
    print('{}')
