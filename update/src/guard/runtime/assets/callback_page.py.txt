"""Read one immutable file prefix. Stdlib only; never delete or acknowledge data."""
import base64
import hashlib
import json
import os
from pathlib import Path
import re
import stat
import sys
import time

MAX_PAGE = 384 * 1024
MAX_ROW = 256 * 1024
NAME = re.compile(r'callbacks-([1-9][0-9]*)(?:-[0-9]{4})?\.jsonl')


TERMINAL = re.compile(r'callbacks-([1-9][0-9]*)\.terminal\.json')
SHA256 = re.compile('[0-9a-f]{64}')
MAX_TERMINAL_BYTES = 4096
MAX_TERMINAL_READ = 32 * 1024**2


def _start_ticks(pid):
    try:
        return int(Path('/proc/' + str(pid) + '/stat').read_text().rsplit(')', 1)[1].split()[19])
    except FileNotFoundError:
        return None


def _terminal_bytes(path, limit, *, expected=None):
    def identity(info):
        return (info.st_dev, info.st_ino, info.st_size, info.st_mtime_ns, info.st_ctime_ns, info.st_nlink)

    before = path.lstat()
    if (not stat.S_ISREG(before.st_mode) or before.st_nlink != 1 or not 0 < before.st_size <= limit
            or expected is not None and (before.st_dev, before.st_ino, before.st_size)
            != (expected['device'], expected['inode'], expected['size'])):
        raise ValueError('CALLBACK_TERMINAL_INCOMPLETE')
    stamp = identity(before)
    with os.fdopen(os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK), 'rb') as stream:
        if identity(os.fstat(stream.fileno())) != stamp:
            raise ValueError('CALLBACK_TERMINAL_INCOMPLETE')
        raw = stream.read(before.st_size + 1)
        if len(raw) != before.st_size or identity(os.fstat(stream.fileno())) != stamp or identity(path.lstat()) != stamp:
            raise ValueError('CALLBACK_TERMINAL_INCOMPLETE')
    return raw


def _terminal_packets(root, names, files, snapshot, start_cursor, session):
    candidates = [(name, int(match[1])) for name in sorted(names) if (match := TERMINAL.fullmatch(name))]
    if not candidates:
        return []
    session_sha256 = hashlib.sha256(session.encode()).hexdigest()
    packets, remaining = [], MAX_TERMINAL_READ
    for name, pid in candidates[:8]:
        expected = {file: item for file, item in snapshot.items() if item['pid'] == pid}
        if (not 1 <= len(expected) <= 32 or set(expected) != {file for file in files if int(NAME.fullmatch(file)[1]) == pid}
                or any(type(item['size']) is not int or not 0 < item['size'] <= 1024**2
                       or start_cursor.get(file, 0) != 0 for file, item in expected.items())):
            continue
        total = sum(item['size'] for item in expected.values())
        if total > min(remaining, 16 * 1024**2):
            continue
        remaining -= total
        try:
            raw = _terminal_bytes(root / name, MAX_TERMINAL_BYTES)
            marker = json.loads(raw)
            keys = {'schema', 'pid', 'start_ticks', 'process_sha256', 'session_sha256',
                    'accepted', 'processed', 'written', 'segments'}
            if (type(marker) is not dict or marker.keys() != keys
                    or raw != (json.dumps(marker, sort_keys=True, separators=(',', ':')) + '\n').encode()
                    or marker['schema'] != 'guard-observer-journal-terminal/1'
                    or type(marker['pid']) is not int or marker['pid'] != pid
                    or type(marker['start_ticks']) is not int or marker['start_ticks'] <= 0
                    or marker['session_sha256'] != session_sha256
                    or type(marker['process_sha256']) is not str or not SHA256.fullmatch(marker['process_sha256'])
                    or any(type(marker[key]) is not int or marker[key] <= 0 for key in ('accepted', 'processed', 'written'))
                    or not marker['accepted'] == marker['processed'] == marker['written']
                    or type(marker['segments']) is not list or len(marker['segments']) != len(expected)):
                continue
            ticks = marker['start_ticks']
            if (_start_ticks(os.getpid()) is None or _start_ticks(pid) not in (None, ticks)
                    or any(item['start_ticks'] not in (None, ticks) for item in expected.values())):
                continue
            state_path = root / f'callbacks-{pid}.state.json'
            state_raw = _terminal_bytes(state_path, MAX_TERMINAL_BYTES)
            state = json.loads(state_raw)
            if (type(state) is not dict or state.get('schema') != 'guard-observer-journal/3'
                    or state.get('closed') is not True
                    or any(type(state.get(key)) is not int or state[key] != marker[key]
                           for key in ('accepted', 'processed', 'written'))
                    or any(type(state.get(key)) is not int or state[key] != 0 for key in ('dropped', 'errors'))):
                continue
            count = 0
            for index, segment in enumerate(marker['segments']):
                file = f'callbacks-{pid}-{index:04d}.jsonl'
                if (file not in expected or type(segment) is not dict or segment.keys() != {'name', 'bytes', 'sha256'}
                        or segment['name'] != file or type(segment['bytes']) is not int
                        or segment['bytes'] != expected[file]['size']):
                    raise ValueError('CALLBACK_TERMINAL_INCOMPLETE')
                data = _terminal_bytes(root / file, segment['bytes'], expected=expected[file])
                if not data.endswith(b'\n') or hashlib.sha256(data).hexdigest() != segment['sha256']:
                    raise ValueError('CALLBACK_TERMINAL_INCOMPLETE')
                count += data.count(b'\n')
            if (count != marker['written'] or _terminal_bytes(state_path, MAX_TERMINAL_BYTES) != state_raw
                    or _terminal_bytes(root / name, MAX_TERMINAL_BYTES) != raw):
                continue
        except (OSError, ValueError, IndexError, RecursionError):
            continue
        for item in expected.values():
            item['start_ticks'] = ticks
        packets.append({'raw_sha256': hashlib.sha256(raw).hexdigest(), 'raw_b64': base64.b64encode(raw).decode('ascii')})
    return packets


def page(root, snapshot=None, cursor=None, capture_id=None, seek_end=False, *, session=None, terminal_epochs=None):
    root = Path(root)
    if root.is_symlink() or not root.is_dir():
        raise ValueError('CALLBACK_ROOT_INVALID')
    names = [p.name for p in root.iterdir()]
    if terminal_epochs is not None:
        if (type(terminal_epochs) is not dict or not 1 <= len(terminal_epochs) <= 8 or seek_end
                or type(session) is not str or not re.fullmatch('[0-9a-f]{32}', session)
                or type(capture_id) is not str or not re.fullmatch('[0-9a-f]{32}', capture_id)
                or any(type(pid) is not str or not re.fullmatch('[1-9][0-9]{0,9}', pid)
                       or type(ticks) is not int or ticks <= 0 or _start_ticks(int(pid)) != ticks
                       for pid, ticks in terminal_epochs.items())
                or (snapshot is None and cursor)):
            raise ValueError('CALLBACK_WORKER_TERMINAL_SCOPE_INVALID')
        writer_name = re.compile(r'callbacks-([1-9][0-9]*)(?:-[0-9]{4}\.jsonl|\.(?:state|terminal)\.json)')
        names = [name for name in names if (match := writer_name.fullmatch(name)) and match[1] in terminal_epochs]
    files = sorted(name for name in names if NAME.fullmatch(name))
    if len(files) > 256:
        raise ValueError('CALLBACK_PROCESS_LIMIT')
    start_cursor = dict(cursor or {})
    checkpoints, checkpoint_gaps = {}, []
    if terminal_epochs is None and snapshot is None and capture_id is not None:
        if re.fullmatch('[0-9a-f]{32}', capture_id) is None:
            raise ValueError('CALLBACK_CAPTURE_ID_INVALID')
        request = root / 'capture-request.json'
        tmp = root / ('capture-request-' + capture_id + '.tmp')
        with tmp.open('x') as stream: json.dump({'capture_id': capture_id}, stream)
        os.replace(tmp, request)
        pids = {int(m[1]) for name in names if (m := re.fullmatch(r'callbacks-([1-9][0-9]*)\.state\.json', name))}
        until = time.monotonic() + 1
        while pids - checkpoints.keys() and time.monotonic() < until:
            for pid in pids - checkpoints.keys():
                ack = root / f'checkpoint-{pid}-{capture_id}.json'
                if ack.is_file() and not ack.is_symlink():
                    with ack.open('rb') as stream: raw = stream.read(65537)
                    item = json.loads(raw)
                    if len(raw) > 65536 or item.get('capture_id') != capture_id or item.get('pid') != pid or item.get('schema') != 'guard-observer-checkpoint/1':
                        raise ValueError('CALLBACK_CHECKPOINT_INVALID')
                    checkpoints[pid] = item
            if pids - checkpoints.keys(): time.sleep(0.01)
        if pids - checkpoints.keys(): checkpoint_gaps.append('CALLBACK_CHECKPOINT_UNAVAILABLE')
    elif terminal_epochs is None and capture_id is not None:
        for pid in {v['pid'] for v in snapshot.values()}:
            ack = root / f'checkpoint-{pid}-{capture_id}.json'
            if ack.is_file() and not ack.is_symlink():
                with ack.open('rb') as stream: raw = stream.read(65537)
                item = json.loads(raw)
                if len(raw) > 65536 or item.get('capture_id') != capture_id or item.get('pid') != pid:
                    raise ValueError('CALLBACK_CHECKPOINT_INVALID')
                checkpoints[pid] = item
    terminals = []
    first = snapshot is None
    if first:
        if terminal_epochs is None:
            names = [p.name for p in root.iterdir()]
            files = sorted(name for name in names if NAME.fullmatch(name))
        snapshot = {}
        for name in files:
            info = (root / name).lstat()
            if not stat.S_ISREG(info.st_mode):
                raise ValueError('CALLBACK_FILE_INVALID')
            pid = int(NAME.fullmatch(name)[1])
            if pid in checkpoints and name not in checkpoints[pid]['offsets']:
                continue  # A segment born after the acknowledged cut.
            try:
                ticks = _start_ticks(pid)
            except (OSError, ValueError, IndexError):
                ticks = None
            snapshot[name] = dict(inode=info.st_ino, device=info.st_dev, size=info.st_size, pid=pid, start_ticks=ticks)
            if pid in checkpoints:
                snapshot[name]['size'] = checkpoints[pid]['offsets'].get(name, 0)
        if terminal_epochs is None and capture_id is not None and not seek_end and session is not None:
            terminals = _terminal_packets(root, names, files, snapshot, start_cursor, session)
    if terminal_epochs is not None:
        if (set(files) != set(snapshot)
                or any(str(item['pid']) not in terminal_epochs or item['start_ticks'] != terminal_epochs[str(item['pid'])]
                       for item in snapshot.values())):
            raise ValueError('CALLBACK_WORKER_TERMINAL_SCOPE_INVALID')
        packets = _terminal_packets(root, names, files, snapshot, {}, session)
        if len(packets) != len(terminal_epochs):
            checkpoint_gaps.append('CALLBACK_WORKER_TERMINAL_UNVERIFIED')
        elif first:
            terminals = packets

    def result(cursor, rows, gaps, done):
        value = dict(schema='guard-callback-page/2', snapshot=snapshot, cursor=cursor,
                     start_cursor=start_cursor, rows=rows, gaps=gaps, done=done, capture_id=capture_id)
        if terminals:
            value['journal_terminals'] = terminals
        return value

    cursor = dict(cursor or {})
    rows, size, gaps = [], 0, list(checkpoint_gaps)
    for pid in ({item['pid'] for item in snapshot.values()} | {int(found[1]) for name in names
                if (found := re.fullmatch(r'callbacks-([1-9][0-9]*)\.state\.json', name))}):
        path = root / f'callbacks-{pid}.state.json'
        segmented = any(name.startswith(f'callbacks-{pid}-') for name in snapshot)
        if path.exists():
            if path.is_symlink():
                raise ValueError('CALLBACK_STATE_INVALID')
            with os.fdopen(os.open(path, os.O_RDONLY | os.O_NOFOLLOW), 'rb') as stream:
                raw = stream.read(4097)
            if len(raw) > 4096:
                raise ValueError('CALLBACK_STATE_INVALID')
            state = json.loads(raw)
            if pid in checkpoints:
                state = checkpoints[pid]
            if (state.get('schema') not in {'guard-observer-journal/2', 'guard-observer-journal/3', 'guard-observer-checkpoint/1'}
                    or any(type(state.get(k)) is not int or state[k] < 0 for k in ('accepted', 'written', 'dropped', 'errors'))
                    or pid not in checkpoints and type(state.get('closed')) is not bool or state.get('dropped') != 0
                    or state.get('errors') != 0 or state.get('accepted') != state.get('written')):
                gaps.append('CALLBACK_JOURNAL_INCOMPLETE')
            if pid not in checkpoints and state.get('closed') is not True:
                gaps.append('CALLBACK_CAPTURE_NOT_FINALIZED')
        elif segmented:
            gaps.append('CALLBACK_JOURNAL_STATE_UNAVAILABLE')
    if (not checkpoints and files != sorted(snapshot)) or any(name not in files for name in snapshot):
        gaps.append('CALLBACK_SOURCE_SET_CHANGED')
    if terminal_epochs is None and capture_id is not None and any(item['pid'] not in checkpoints for item in snapshot.values()):
        gaps.append('CALLBACK_CHECKPOINT_UNAVAILABLE')
    if seek_end:
        return result({name: item['size'] for name, item in snapshot.items()}, [], gaps, True)
    for name, expected in sorted(snapshot.items()):
        if NAME.fullmatch(name) is None:
            raise ValueError('CALLBACK_CURSOR_INVALID')
        offset = cursor.get(name, 0)
        if type(offset) is not int or not 0 <= offset <= expected['size']:
            raise ValueError('CALLBACK_CURSOR_INVALID')
        fd = os.open(root / name, os.O_RDONLY | os.O_NOFOLLOW)
        with os.fdopen(fd, 'rb') as stream:
            info = os.fstat(stream.fileno())
            if (not stat.S_ISREG(info.st_mode) or info.st_ino != expected['inode'] or info.st_dev != expected['device']
                    or info.st_size < expected['size']):
                raise ValueError('CALLBACK_FILE_REPLACED')
            stream.seek(offset)
            while stream.tell() < expected['size']:
                begin = stream.tell()
                raw = stream.readline(min(MAX_ROW + 1, expected['size'] - begin))
                if len(raw) > MAX_ROW or not raw.endswith(b'\n'):
                    gaps.append('CALLBACK_ROW_INCOMPLETE')
                    cursor[name] = expected['size']
                    break
                if size + len(raw) > MAX_PAGE:
                    cursor[name] = begin
                    return result(cursor, rows, gaps, False)
                value = json.loads(raw)
                rows.append(dict(file=name, offset=begin, end=stream.tell(), raw_sha256=hashlib.sha256(raw).hexdigest(), event=value))
                size += len(raw)
                cursor[name] = stream.tell()
    return result(cursor, rows, gaps, True)


def container_start():
    try:
        import time
        ticks = int(Path('/proc/1/stat').read_text().rsplit(')', 1)[1].split()[19])
        offset = time.clock_gettime_ns(time.CLOCK_BOOTTIME) - time.monotonic_ns()
        return ticks * 1000000000 // os.sysconf('SC_CLK_TCK') - offset
    except (OSError, ValueError, IndexError, AttributeError):
        return None


if __name__ == '__main__':
    spec = json.loads(sys.argv[1])
    if re.fullmatch('[0-9a-f]{32}', spec['session']) is None:
        raise ValueError('CALLBACK_SESSION_INVALID')
    value = page('/guard-lifecycle/data/' + spec['session'], spec.get('snapshot'), spec.get('cursor'),
                 spec.get('capture_id'), spec.get('seek_end', False), session=spec['session'])
    value['container_start_monotonic_ns'] = container_start()
    print(json.dumps(value, separators=(',', ':')))
