"""Passive owned-child proc/cgroup evidence around the unchanged SUP15 wait."""
from pathlib import Path
import hashlib,json,os,threading,time,uuid

SHA=lambda p:hashlib.sha256(Path(p).read_bytes()).hexdigest()

def read_at(fd,name,bound):
    child=os.open(name,os.O_RDONLY|os.O_NOFOLLOW,dir_fd=fd)
    with os.fdopen(child,'rb') as stream:raw=stream.read(bound+1)
    if len(raw)>bound:raise ValueError('SUP15_OBSERVER_READ_BOUND:'+name)
    return raw.decode(errors='replace')

def identity(fd):
    raw=read_at(fd,'stat',4096);fields=raw.rsplit(')',1)[1].split()
    status=read_at(fd,'status',8192)
    uids=[int(x) for x in next(line for line in status.splitlines() if line.startswith('Uid:')).split()[1:]]
    return {'PID':int(raw.split('(',1)[0]),'PPID':int(fields[1]),'PGID':int(fields[2]),'SID':int(fields[3]),
            'start_ticks':int(fields[19]),'uids':uids},raw,status

def sample_owned(fd,binding):
    before,stat,status=identity(fd)
    expected={k:binding[k] for k in ['PID','PPID','PGID','SID','start_ticks']}
    if any(before[k]!=v for k,v in expected.items()) or before['uids']!=[1000]*4:raise ValueError('SUP15_OBSERVER_SUBJECT_CHANGED')
    row={'monotonic_ns':time.monotonic_ns(),'binding':expected,'stat':stat,'status':status,'proc':{},'cgroup':{},'cgroup_scope':'whole container incl observer/other children; not isolated subject CPU attribution'}
    for name,bound in [('io',4096),('schedstat',1024),('wchan',1024),('cgroup',4096)]:
        try:row['proc'][name]={'raw':read_at(fd,name,bound),'error_type':None}
        except (OSError,ValueError) as exc:row['proc'][name]={'raw':None,'error_type':type(exc).__name__}
    cgroup=row['proc']['cgroup'].get('raw')
    if cgroup is not None and cgroup==Path('/proc/self/cgroup').read_text():
        for name in ['cpu.max','cpu.stat','memory.current','memory.events']:
            try:
                with (Path('/sys/fs/cgroup')/name).open('rb') as stream:raw=stream.read(4097)
                if len(raw)>4096:raise ValueError('SUP15_CGROUP_BOUND')
                row['cgroup'][name]={'raw':raw.decode(),'error_type':None}
            except (OSError,ValueError) as exc:row['cgroup'][name]={'raw':None,'error_type':type(exc).__name__}
    else:row['cgroup_unavailable_reason']='subject_and_observer_cgroup_binding_not_equal'
    after,_,_=identity(fd)
    if after!=before:raise ValueError('SUP15_OBSERVER_SUBJECT_CHANGED_AFTER_READ')
    return row

def observe_one(*,original_wait,argv,env,output,stem,limit,bootstrap,allowed_source_SHA,interval=1.0,reader=sample_owned):
    """Preparation API only: fixed source targets, original wait/cleanup function."""
    if os.getuid()!=1000 or len(argv)<4 or argv[1:3]!=['-I','-B']:raise ValueError('SUP15_OBSERVER_EXACT_ARGV_REQUIRED')
    target=Path(argv[3])
    if SHA(target) not in allowed_source_SHA:raise ValueError('SUP15_OBSERVER_TARGET_UNSUPPORTED')
    if not 0<limit<=90 or not 0<interval<=1:raise ValueError('SUP15_OBSERVER_BOUNDS_CHANGED')
    diagnostic=output/(stem+'-diagnostics');diagnostic.mkdir(mode=0o700)
    journal=diagnostic/'phases.jsonl';stack=diagnostic/'before-deadline-stack.log';nonce=uuid.uuid4().hex
    config={'nonce':nonce,'bootstrap_SHA':SHA(bootstrap),'target':str(target),'target_SHA':SHA(target),'target_args':argv[4:],
            'journal':str(journal),'stack':str(stack),'stack_after_seconds':85 if limit==90 else 40 if limit==45 else max(.05,limit-5)}
    config_file=diagnostic/'config.json';config_file.write_text(json.dumps(config,indent=2),encoding='utf8')
    wrapped=[argv[0],'-I','-B',str(bootstrap),'--config',str(config_file),'--config-sha',SHA(config_file)]
    done=threading.Event();observations=[];errors=[];binding_box=[];cpu_start=time.process_time_ns();fd_box=[]
    def watch():
        fd=None
        try:
            with (diagnostic/'resources.jsonl').open('x',encoding='utf8') as stream:
                def emit(value):
                    raw=json.dumps(value,separators=(',',':'))+'\n'
                    if stream.tell()+len(raw.encode())>2*1024**2:raise ValueError('SUP15_RESOURCE_JOURNAL_BOUND')
                    stream.write(raw);stream.flush();os.fsync(stream.fileno())
                while not done.is_set() and not binding_box:
                    if journal.exists():
                        raw=journal.read_bytes()[:65536];line=raw.splitlines(keepends=True)[:1]
                        if line and line[0].endswith(b'\n'):
                            b=json.loads(line[0]);expected_ppid=os.getpid()
                            if (b['nonce']!=nonce or b['uid']!=1000 or b['PPID']!=expected_ppid or b['PGID']!=b['PID'] or b['SID']!=b['PID'] or b['target_SHA']!=config['target_SHA']):
                                raise ValueError('SUP15_OBSERVER_CHILD_BINDING_UNVERIFIED')
                            fd=os.open('/proc/'+str(b['PID']),os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW);fd_box.append(fd)
                            first=reader(fd,b);binding_box.append(b);observations.append(first);emit(first)
                            break
                    done.wait(.01)
                while binding_box and not done.wait(interval):
                    row=reader(fd,binding_box[0]);observations.append(row);emit(row)
                    if len(observations)>=128:raise ValueError('SUP15_RESOURCE_SAMPLE_BOUND')
        except BaseException as exc:errors.append({'type':type(exc).__name__,'message':str(exc)})
        finally:
            if fd is not None:os.close(fd)
    worker=threading.Thread(target=watch,name='sup15-owned-resource-observer');worker.start()
    primary=None;rc=None;termination=None
    try:rc,termination=original_wait(wrapped,env,output,stem,limit)
    except BaseException as exc:primary={'type':type(exc).__name__,'message':str(exc)};raise
    finally:
        done.set();worker.join(2)
        summary={'original_argv':argv,'observed_argv':wrapped,'original_limit':limit,'original_wait_return':rc,'original_termination':termination,
                 'original_primary':primary,'binding':binding_box[0] if binding_box else None,'samples':len(observations),'observer_errors':errors,
                 'observer_thread_stopped':not worker.is_alive(),'observer_parent_CPU_ns_including_its_controller':time.process_time_ns()-cpu_start,
                 'observer_influence':'extra parent sampling thread, child daemon Python stack timer and own atexit markers; GIL scheduling can delay/miss stack; no performance equivalence claim',
                 'cgroup_scope':'container total, includes observer','target_runtime_model_or_product_claim':False}
        if termination and binding_box and termination['PID']!=binding_box[0]['PID']:summary['observer_errors'].append({'type':'ValueError','message':'SUP15_WAIT_PID_BINDING_MISMATCH'})
        with (diagnostic/'summary.json').open('x',encoding='utf8') as stream:json.dump(summary,stream,indent=2)
    return rc,termination,summary
