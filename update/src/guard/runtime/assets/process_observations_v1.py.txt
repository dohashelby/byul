"""Fixed Linux process metadata reader. No argv, file contents or env values out."""
import datetime
import hashlib
import json
import math
import os
import re
import stat
import sys
import threading

SCHEMA = 'guard-container-process-observations/1'
MAX_PIDS = 128
MAX_NAMES = 64
MAX_OUTPUT = 524288


def stamp():
    return datetime.datetime.now(datetime.timezone.utc).isoformat()


def bounded(path, limit):
    with open(path, 'rb') as stream:
        raw = stream.read(limit + 1)
    if len(raw) > limit:
        raise ValueError('bound')
    return raw


def proc_start(pid):
    raw = bounded('/proc/'+str(pid)+'/stat', 8192).decode('utf-8')
    return int(raw.rsplit(')', 1)[1].split()[19])


def namespace(pid, kind):
    value = os.readlink('/proc/'+str(pid)+'/ns/'+kind)
    match = re.fullmatch(kind+r':\[([1-9][0-9]{0,19})\]', value)
    if match is None:
        raise ValueError('namespace')
    return int(match.group(1))


def process(pid):
    row = {'pid': pid, 'start_ticks': None, 'name': None, 'uids': None,
           'uid_effective': None, 'cap_eff': None, 'mount_namespace_inode': None,
           'environment_names': [], 'environment_names_complete': False,
           'pid_identity_stable': False, 'started_at': stamp(), 'finished_at': None,
           'diagnostic_codes': []}
    try:
        row['start_ticks'] = proc_start(pid)
    except Exception:
        row['diagnostic_codes'].append('PROCESS_IDENTITY_UNAVAILABLE')
        row['finished_at'] = stamp()
        return row
    try:
        lines = bounded('/proc/'+str(pid)+'/status', 8192).decode('utf-8').splitlines()
        fields = {}
        for line in lines:
            key, sep, value = line.partition(':')
            if key in {'Name','Uid','CapEff'} and sep:
                if key in fields:
                    raise ValueError('duplicate')
                fields[key] = value.strip()
        name, uids, caps = fields['Name'], fields['Uid'].split(), fields['CapEff']
        if (re.fullmatch(r'[A-Za-z0-9_.:/+ -]{1,64}',name) is None
                or len(uids) != 4 or any(re.fullmatch(r'[0-9]{1,10}',x) is None for x in uids)
                or any(int(x) > 4294967295 for x in uids)
                or re.fullmatch(r'[0-9a-fA-F]{1,16}',caps) is None):
            raise ValueError('status')
        row.update(name=name,uids=[int(x) for x in uids],uid_effective=int(uids[1]),cap_eff=caps.lower())
    except Exception:
        row['diagnostic_codes'].append('PROCESS_STATUS_UNAVAILABLE')
    try:
        row['mount_namespace_inode'] = namespace(pid,'mnt')
    except Exception:
        row['diagnostic_codes'].append('PROCESS_MOUNT_NAMESPACE_UNAVAILABLE')
    try:
        # Values exist only transiently in this reader. Names are the sole output.
        raw = bounded('/proc/'+str(pid)+'/environ',65536)
        complete = not raw or raw.endswith(bytes([0]))
        names = set()
        for item in raw.split(bytes([0])):
            if not item:
                continue
            name, sep, _ = item.partition(b'=')
            if not sep or re.fullmatch(rb'[A-Za-z_][A-Za-z0-9_]{0,127}',name) is None:
                complete = False
                continue
            names.add(name.decode('ascii'))
        if len(names) > MAX_NAMES:
            complete = False
        row['environment_names'] = sorted(names)[:MAX_NAMES]
        row['environment_names_complete'] = complete
        if not complete:
            row['diagnostic_codes'].append('PROCESS_ENVIRONMENT_NAMES_INCOMPLETE')
    except Exception:
        row['diagnostic_codes'].append('PROCESS_ENVIRONMENT_NAMES_UNAVAILABLE')
    try:
        row['pid_identity_stable'] = proc_start(pid) == row['start_ticks']
        if not row['pid_identity_stable']:
            row['diagnostic_codes'].append('PROCESS_IDENTITY_CHANGED')
    except Exception:
        row['diagnostic_codes'].append('PROCESS_IDENTITY_CHANGED')
    row['finished_at'] = stamp()
    return row


def mark(value):
    return {'device':value.st_dev,'inode':value.st_ino,'mode':value.st_mode,
            'uid':value.st_uid,'gid':value.st_gid,'size':value.st_size,
            'mtime_ns':value.st_mtime_ns,'ctime_ns':value.st_ctime_ns}


def metadata(path):
    # Parent directories are opened without following links. Candidate file
    # contents and symlink targets are never opened or read.
    fd = os.open('/',os.O_PATH|os.O_DIRECTORY)
    chain = []
    try:
        parts = path.strip('/').split('/')
        if parts == ['']:
            return 'unavailable',None,'DIRECTORY_SCOPE_UNVERIFIED',chain
        for index,part in enumerate(parts):
            try:
                info = os.stat(part,dir_fd=fd,follow_symlinks=False)
            except FileNotFoundError:
                return 'absent',None,'PATH_ABSENT',chain
            except PermissionError:
                return 'unavailable',None,'PATH_METADATA_UNAVAILABLE',chain
            chain.append(mark(info))
            if stat.S_ISLNK(info.st_mode):
                return 'symlink_not_followed',mark(info),'SYMLINK_NOT_FOLLOWED',chain
            if index < len(parts)-1:
                if not stat.S_ISDIR(info.st_mode):
                    return 'absent',None,'PARENT_NOT_DIRECTORY',chain
                child = os.open(part,os.O_PATH|os.O_DIRECTORY|os.O_NOFOLLOW,dir_fd=fd)
                if mark(os.fstat(child)) != mark(info):
                    os.close(child)
                    return 'unavailable',None,'PATH_IDENTITY_CHANGED',chain
                os.close(fd)
                fd = child
            else:
                return 'metadata',mark(info),'LSTAT_METADATA_ONLY',chain
    finally:
        os.close(fd)


def path_record(path):
    row = {'path':path,'state':'unavailable','metadata':None,'metadata_identity_stable':False,
           'path_chain_sha256':None,'access_tested':False,'started_at':stamp(),'finished_at':None,
           'diagnostic_codes':[]}
    try:
        first = metadata(path)
        row.update(state=first[0],metadata=first[1],
                   path_chain_sha256=hashlib.sha256(json.dumps(first[3],sort_keys=True).encode()).hexdigest())
        second = metadata(path)
        row['metadata_identity_stable'] = first == second
        if first[2] != 'LSTAT_METADATA_ONLY':
            row['diagnostic_codes'].append(first[2])
        if first != second:
            row['diagnostic_codes'].append('PATH_IDENTITY_CHANGED')
    except Exception:
        row['diagnostic_codes'].append('PATH_METADATA_UNAVAILABLE')
    row['finished_at'] = stamp()
    return row


def pids_excluding(capture_pid):
    found = []
    with os.scandir('/proc') as entries:
        for entry in entries:
            if entry.name.isdigit() and int(entry.name) != capture_pid:
                found.append(int(entry.name))
                # The extra item is an explicit lower-bound marker. Directory
                # enumeration itself stays bounded, not just later file reads.
                if len(found) > MAX_PIDS:
                    break
    return sorted(found)


def capture(paths):
    begin = stamp()
    capture_pid = os.getpid()
    pid_ns = namespace(capture_pid,'pid')
    mount_ns = namespace(capture_pid,'mnt')
    own_uid = os.geteuid()
    before = pids_excluding(capture_pid)
    selected = before[:MAX_PIDS]
    errors = [] if len(before) <= MAX_PIDS else ['PROCESS_SET_BOUND']
    if not before:
        errors.append('PROCESS_SET_EMPTY')
    rows = [process(pid) for pid in selected]
    candidates = [path_record(path) for path in paths]
    after = pids_excluding(capture_pid)
    observer_stable = os.geteuid() == own_uid and namespace(capture_pid,'pid') == pid_ns and namespace(capture_pid,'mnt') == mount_ns
    if before != after:
        errors.append('PROCESS_SET_CHANGED')
    if len(after) > MAX_PIDS:
        errors.append('PROCESS_SET_BOUND')
    if not after:
        errors.append('PROCESS_SET_EMPTY')
    if not observer_stable:
        errors.append('CAPTURE_SUBJECT_CHANGED')
    inventory_stable = (bool(before) and len(before) <= MAX_PIDS and before == after and observer_stable
                        and all(x['pid_identity_stable'] for x in rows))
    complete = inventory_stable and all(not x['diagnostic_codes'] for x in rows)
    value = {'schema':SCHEMA,'started_at':begin,'finished_at':stamp(),
        'capture_subject':{'pid':capture_pid,'uid_effective':own_uid,'pid_namespace_inode':pid_ns,
                           'mount_namespace_inode':mount_ns,'identity_stable':observer_stable},
        'capture_process_excluded':True,'processes':rows,'pid_count_before':len(before),
        'pid_count_after':len(after),'pid_inventory_stable':inventory_stable,'process_metadata_complete':complete,
        'candidate_metadata':candidates,'worker_subject_verified':False,'worker_inventory_complete':False,
        'diagnostic_codes':sorted(set(errors))}
    while len(json.dumps(value,separators=(',',':')).encode()) > MAX_OUTPUT and value['processes']:
        value['processes'].pop()
        value['pid_inventory_stable'] = False
        value['process_metadata_complete'] = False
        value['diagnostic_codes'] = sorted(set(value['diagnostic_codes']+['PROCESS_OUTPUT_BOUND']))
    return value


def main():
    budget = float(sys.argv[1])
    if not math.isfinite(budget) or not 0 < budget <= 300:
        raise ValueError('budget')
    watchdog = threading.Timer(budget,lambda:os._exit(124))
    watchdog.daemon = True
    watchdog.start()
    raw = sys.stdin.buffer.read(65537)
    if len(raw) > 65536:
        raise ValueError('bound')
    request = json.loads(raw)
    if type(request) is not dict or set(request) != {'candidate_paths'}:
        raise ValueError('request')
    paths = request['candidate_paths']
    if (type(paths) is not list or len(paths)>32 or any(type(x) is not str for x in paths)
            or len(set(paths))!=len(paths)):
        raise ValueError('paths')
    for path in paths:
        if (not path.startswith('/') or len(path.encode())>128 or '\\' in path or '//' in path
                or any(ord(c)<32 or ord(c)==127 for c in path)
                or any(x in {'.','..'} for x in path.split('/'))
                or len(path)>1 and path.endswith('/')):
            raise ValueError('path')
    value = capture(paths)
    encoded = json.dumps(value,separators=(',',':')).encode()
    if len(encoded) > MAX_OUTPUT:
        raise ValueError('output')
    sys.stdout.buffer.write(encoded+b'\n')


if __name__ == '__main__':
    try:
        main()
    except Exception:
        sys.stdout.write(json.dumps({'schema':SCHEMA,'program_error':'PROCESS_PROGRAM_UNAVAILABLE'})+'\n')
        raise SystemExit(1) from None
